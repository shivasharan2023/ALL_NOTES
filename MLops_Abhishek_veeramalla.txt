##deployment.yml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: intent-classifier
  namespace: intent-namespace
  labels:
    app: intent-classifier
spec:
  replicas: 2
  selector:
    matchLabels:
      app: intent-classifier
  template:				# this is for labelling pods
    metadata:
      labels:
        app: intent-classifier
    spec:
      containers:
        - name: intent-classifier
          image: <image-name>
          ports:
            - containerPort: 6000

### INSTALL KSERVE
#install cert manager
#should have cert manager installed
>kubectl apply -f https://github.com/cert-manager/releases/latest/download/cert-manager.yaml
>kubectl get pods -n cert-manager

# install CRDS (Custom Resource Definition)
>kubectl create namespace kserve		#create namespace
>helm install kserve-crd oci://ghcr.io/kserve/charts/kserve-crd \		#installs kserve-crd

# install kserve
>helm install kserve oci://ghcr.io/kserve/charts/kserve \		#installs kserve #kserve installed at cluster level

# attach model as release binary to github repo
github repo>>select branch >> tag>>releases>>Draft a new release>>Traget-Kserve>>select tag>>create new tag - 3.0>>create>>release title - v3.0>>attach binaries by dropping them - attach the .pkl model>>publish release 

github repo>>select branch >>KServe-implementation.md>> change storageUri to 

# create namespace
>kubectl create ns intent 
>cat <<EOF | kubectl apply -n intent -f -
apiVersion: serving.kserve.io/v1beta1
kind: InferenceService
metadata:
  name: intent-classifier
spec:
  predictor:
    model:
      modelFormat:
        name: sklearn
      storageUri: "<downloadable location>"		# Change this something like https://github.com/iam-veeramalla/Intent-classifier-model/releases/donwload/3.0/intent_model.pkl
      resources:
        requests:
          cpu: "100m"
          memory: "512Mi"
        limits:
          cpu: "1"
          memory: "1Gi"
EOF

# View kserve logs
>kubectl get pods -n kserve				# copy name of the pod
>kubectl logs <name_of_kserve_pod> -n kserve

# Confirm horizontalpodautoscaler created
>kubectl get horizontalpodautoscale.autoscaling -n intent 

# Confirm service created
>kubectl get svc -n intent

# Enable port forwarding
>kubectl port-forward scv/<name_of_svc> 8080:80	-n intent --address 0.0.0.0		#80:service port   8080:localport
